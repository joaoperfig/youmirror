# youmirror – Python dependencies
#
# IMPORTANT: picamera2 is NOT listed below and must NOT be pip-installed.
# It requires libcamera's compiled Python bindings, which pip can only get by
# building PyAV/python-prctl from source — this needs FFmpeg dev headers and
# is slow/fragile on a Pi Zero W. Install it via apt instead (see below),
# and create your venv with --system-site-packages so it can see it.
#
#   sudo apt-get update
#   sudo apt-get install -y python3-pip python3-dev
#   sudo apt-get install -y python3-smbus i2c-tools
#   sudo apt-get install -y python3-picamera2
#
#   python3 -m venv --system-site-packages venv
#   source venv/bin/activate
#   pip install -r requirements.txt
#
# The legacy `picamera` library is NOT used in this project: on this Pi's
# kernel/firmware, the legacy MMAL camera stack fails to detect the OV5647
# sensor (vcgencmd get_camera reports detected=0) even though the sensor is
# fine — libcamera/picamera2 detects and drives it correctly.
#
# On armv6l (Pi Zero / Pi 1) OpenCV has no usable pip wheel in some cases –
# if `pip install opencv-python-headless` fails, install via apt instead:
#   sudo apt-get install -y python3-opencv

# I2C communication with PCA9685
smbus2

# OpenCV – headless build (no GUI deps) appropriate for a Pi running OS Lite
opencv-python-headless

# NumPy (pulled in by OpenCV but listed explicitly for clarity)
numpy
