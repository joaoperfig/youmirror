# youmirror – Python dependencies
#
# Install on the Raspberry Pi with:
#   pip install -r requirements.txt
#
# Some packages (picamera, RPi.GPIO) are Raspberry Pi–specific and will not
# install on a regular PC. smbus2 and opencv-python-headless will install
# anywhere and can be used for offline development/simulation.
#
# System packages required before pip install (run once on the Pi):
#   sudo apt-get update
#   sudo apt-get install -y python3-pip python3-dev
#   sudo apt-get install -y python3-smbus i2c-tools
#   sudo apt-get install -y python3-picamera   # or via pip below
#
# On armv6l (Pi Zero / Pi 1) OpenCV has no usable pip wheel – install via apt:
#   sudo apt-get install -y python3-opencv
#
# On armv7l / armv8 / x86-64 you can instead let pip install it:
#   pip install opencv-python-headless

# I2C communication with PCA9685
smbus2

# NumPy (pulled in by OpenCV but listed explicitly for clarity)
numpy

# Official Pi Camera library (legacy picamera, for Raspberry Pi OS Legacy)
# This is pre-installed on most Pi OS images; include here for completeness.
picamera
